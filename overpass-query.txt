# v3.2: Vienna U-Bahn + passenger rail only. No tram/light-rail overlay.
# Vienna OSM relation: 109166; Overpass area: 3600109166.

# 1) Vienna boundary + districts
[out:json][timeout:45];
area(3600109166)->.vienna;
(
  relation(109166);
  relation(area.vienna)["boundary"="administrative"]["admin_level"="9"];
);
out geom;

# 2) Selectable stations: U-Bahn and passenger train stations only
[out:json][timeout:45];
area(3600109166)->.vienna;
(
  nwr(area.vienna)["railway"~"^(station|halt)$"]["station"="subway"]["access"!="private"];
  nwr(area.vienna)["railway"~"^(station|halt)$"]["subway"="yes"]["access"!="private"];
  nwr(area.vienna)["railway"~"^(station|halt)$"]["train"="yes"]["tram"!="yes"]["access"!="private"];
  nwr(area.vienna)["public_transport"="station"]["train"="yes"]["tram"!="yes"]["access"!="private"];
);
out center tags;

# The web app collapses same-named OSM station elements to ONE selectable marker.

# 3) Transit line overlay: subway + ordinary rail; explicitly no tram/light_rail
[out:json][timeout:60];
area(3600109166)->.vienna;
(
  way(area.vienna)["railway"="subway"];
  way(area.vienna)["railway"="rail"]["service"!="yard"]["service"!="siding"]["service"!="spur"];
);
out geom tags;
